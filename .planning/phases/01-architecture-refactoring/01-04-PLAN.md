    ```typescript
    /**
     * useSiteSettings — unified frontend settings access
     * Per ARCH-02 / D-05: replaces all manual fetch('/api/settings') calls
     * Uses useState for SSR → client hydration consistency
     */
    export function useSiteSettings() {
      const settings = useState<Record<string, unknown>>('site-settings', () => ({}))
      const loading = useState<boolean>('site-settings-loading', () => false)
      const initialized = useState<boolean>('site-settings-init', () => false)

      async function refresh(): Promise<void> {
        // Skip fetch on server — Nuxt SSR will handle initial data load
        // On client, always fetch fresh data
        if (import.meta.server && initialized.value) return

        loading.value = true
        try {
          const response = await $fetch<{
            code: number
            data: Record<string, Array<{ key: string; value: unknown }>>
          }>('/api/settings')

          if (response.code === 0 && response.data) {
            // Flatten grouped settings into a key-value map
            const flat: Record<string, unknown> = {}
            for (const rows of Object.values(response.data)) {
              for (const row of rows) {
                flat[row.key] = row.value
              }
            }
            settings.value = flat
          }
        } catch (error) {
          console.error('[useSiteSettings] Failed to fetch settings:', error)
        } finally {
          loading.value = false
          initialized.value = true
        }
      }

      // Auto-fetch on first call (client-side only to avoid double fetch with SSR)
      if (import.meta.client && !initialized.value) {
        refresh()
      }

      return {
        settings: readonly(settings),
        loading: readonly(loading),
        refresh,
      }
    }
    ```

    Design decisions per D-05:
    - `useState` keys: `'site-settings'`, `'site-settings-loading'`, `'site-settings-init'` — unique, won't collide with other composables
    - `settings` is `readonly()` to prevent accidental mutation (saving is the admin's job)
    - `refresh()` is exposed for pages that need to re-fetch after a known update (e.g., after admin saves)
    - `import.meta.client` guard prevents client-side double-fetch when data is already SSR-rendered
    - No `useAsyncData()` — `useState` + `$fetch` pattern keeps it simple, matching `useTheme` conventions
    - `SiteSettings` type is inferred as `Record<string, unknown>` — individual pages cast specific keys as needed (e.g., `settings.value.themeConfig as ThemeConfig`)

    Do NOT add type exports — the settings shape is dynamic (any key can exist). Pages cast individual fields.
    Do NOT add auth handling — this is a public endpoint (system settings are public for frontend rendering).

    ```typescript
    import { useAuthStore } from '~/stores/auth'

    /**
     * useAdminSettings — unified admin settings fetch/save
     * Per ARCH-03 / D-06: replaces ~50 lines of duplicated fetch/save logic in each admin page
     * Category parameter scopes to one admin page's settings group
     *
     * Usage: const { settings, loading, save, refresh } = useAdminSettings('general')
     * Then: await save({ darkmode: { enabled: true }, socialLinks: [...] })
     */
    export function useAdminSettings(category: string) {
      const authStore = useAuthStore()
      const settings = ref<Record<string, unknown>>({})
      const loading = ref<boolean>(false)
      const initialized = ref<boolean>(false)

      /** Build auth headers from store */
      function getHeaders(): Record<string, string> {
        const headers: Record<string, string> = {}
        if (authStore.token) {
          headers.Authorization = `Bearer ${authStore.token}`
        }
        return headers
      }

      /** Fetch all settings for this category from the API */
      async function refresh(): Promise<void> {
        loading.value = true
        try {
          const response = await $fetch<{
            code: number
            data: Record<string, Array<{ key: string; value: unknown }>>
          }>('/api/settings', {
            headers: getHeaders(),
          })

          if (response.code === 0 && response.data) {
            const flat: Record<string, unknown> = {}
            for (const rows of Object.values(response.data)) {
              for (const row of rows) {
                flat[row.key] = row.value
              }
            }
            settings.value = flat
          }
        } catch (error: unknown) {
          const err = error as { statusCode?: number; data?: { message?: string } }
          if (err?.statusCode === 401) {
            console.error('[useAdminSettings] Unauthorized — token may be expired')
          }
          console.error(`[useAdminSettings] Failed to fetch settings for category "${category}":`, error)
        } finally {
          loading.value = false
          initialized.value = true
        }
      }

      /**
       * Save settings values to the API
       * Accepts a flat key-value object — each key is written as a separate setting entry
       * with the composable's category
       */
      async function save(values: Record<string, unknown>): Promise<void> {
        const items = Object.entries(values).map(([key, value]) => ({
          key,
          value,
          category,
        }))

        await $fetch('/api/settings', {
          method: 'PUT',
          headers: getHeaders(),
          body: items,
        })

        // Update local state after successful save
        for (const [key, value] of Object.entries(values)) {
          settings.value[key] = value
        }
      }

      // Auto-fetch on first call
      if (!initialized.value) {
        refresh()
      }

      return {
        settings,
        loading: readonly(loading),
        save,
        refresh,
      }
    }
    ```

    Design decisions per D-06:
    - `category` parameter is the composable's scope — used when saving to tag all keys with the same category
    - `save(values)` converts a flat `{key: value}` object into the array format expected by `PUT /api/settings`
    - After save, updates `settings.value` locally — no need to re-fetch from server
    - Uses `useAuthStore()` for token — same auth pattern as `useAdminApi`
    - No `useState` sharing (unlike useSiteSettings) — each admin page gets its own instance. Admin pages are client-side only with no SSR hydration concern
    - Error handling: logs 401 errors (token expiry) but lets the caller handle display

    Do NOT:
    - Import `useAdminApi` — this composable is self-contained, calling `$fetch` directly
    - Add SSR support — admin pages are client-only (behind auth middleware)
    - Export any types beyond what's inferred from the return value
