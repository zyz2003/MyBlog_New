    /**
     * useSiteSettings — frontend composable for site configuration (per ARCH-02, D-05)
     *
     * Returns typed, SSR-safe settings object shared across all frontend pages/components.
     * Uses Nuxt's useState() for SSR/client hydration consistency (same pattern as useTheme.ts).
     *
     * Usage:
     *   const { settings, loading, refresh } = useSiteSettings()
     *   // settings.value.siteName, settings.value.socialLinks, etc.
     *
     * No manual $fetch calls needed in any frontend page or component.
     */

    export function useSiteSettings() {
      const settings = useState<Record<string, unknown>>('siteSettings', () => ({}))
      const loading = ref(true)

      /**
       * Fetch all settings from /api/settings and flatten into a key-value map.
       * API returns settings grouped by category — we flatten to { key: value } for easy access.
       */
      async function refresh(): Promise<void> {
        loading.value = true
        try {
          const data = await $fetch<{
            code: number
            data: Record<string, Array<{ key: string; value: unknown }>>
          }>('/api/settings')

          if (data.code === 0 && data.data) {
            const flat: Record<string, unknown> = {}
            for (const rows of Object.values(data.data)) {
              for (const row of rows) {
                flat[row.key] = row.value
              }
            }
            settings.value = flat
          }
        } catch (e) {
          console.error('[useSiteSettings] Failed to fetch settings:', e)
        } finally {
          loading.value = false
        }
      }

      return { settings, loading, refresh }
    }
    ```

    Design decisions (Claude's Discretion):
    - `settings` is `Ref<Record<string, unknown>>` — keys are dynamic (different admin pages add different keys), so a fixed interface would be incomplete. Consumers access via `settings.value.specificKey` with their own type assertions.
    - No auto-fetch on first call — consumers call `refresh()` explicitly (e.g., in `onMounted` or via `useAsyncData`). This matches the existing pattern (useTheme requires explicit `initTheme()` call).
    - `refresh()` is idempotent — calling it multiple times re-fetches, overwriting local state. Loading flag prevents duplicate concurrent fetches at the caller's discretion.
    - File location: `composables/frontend/useSiteSettings.ts` per D-07.

    /**
     * useAdminSettings — admin composable for settings management (per ARCH-03, D-06)
     *
     * Encapsulates the fetch/save pattern duplicated across 11 admin pages,
     * reducing each page's settings handling from ~50 lines to ~5 lines.
     *
     * Uses Nuxt's useState() for SSR-safe state (same pattern as useTheme.ts).
     *
     * Usage:
     *   const { settings, loading, save, refresh } = useAdminSettings('general')
     *   // settings.value.socialLinks, settings.value.footer, etc.
     *   // save([{ key: 'footer', value: footerData }]) — sends PUT /api/settings
     *
     * @param category - Settings category to scope (e.g., 'general', 'seo', 'display')
     */

    export function useAdminSettings(category: string) {
      const settings = useState<Record<string, unknown>>(
        `admin-settings-${category}`,
        () => ({}),
      )
      const loading = ref(true)

      /**
       * Fetch all settings from /api/settings and flatten into a key-value map.
       * All categories are fetched (API returns everything) and stored in one flat object.
       * The category parameter scopes the useState key to avoid cross-page collisions.
       */
      async function refresh(): Promise<void> {
        loading.value = true
        try {
          const data = await $fetch<{
            code: number
            data: Record<string, Array<{ key: string; value: unknown }>>
          }>('/api/settings')

          if (data.code === 0 && data.data) {
            const flat: Record<string, unknown> = {}
            for (const rows of Object.values(data.data)) {
              for (const row of rows) {
                flat[row.key] = row.value
              }
            }
            settings.value = flat
          }
        } catch (e) {
          console.error(
            `[useAdminSettings] Failed to fetch settings for category "${category}":`,
            e,
          )
        } finally {
          loading.value = false
        }
      }

      /**
       * Save settings values to the server via PUT /api/settings.
       * Each item in the values array must have at least { key, value }.
       * The category field is auto-populated from the composable's category parameter.
       *
       * @param values - Array of setting items to save: [{ key: string, value: unknown }]
       */
      async function save(
        values: Array<{ key: string; value: unknown; category?: string; description?: string }>,
      ): Promise<void> {
        const items = values.map((v) => ({
          key: v.key,
          value: v.value,
          category: v.category || category,
          description: v.description,
        }))

        const data = await $fetch<{ code: number; data: unknown; message: string }>(
          '/api/settings',
          {
            method: 'PUT',
            body: items,
          },
        )

        if (data.code !== 0) {
          throw new Error(data.message || 'Failed to save settings')
        }

        // Update local cache with saved values so UI reflects changes immediately
        for (const item of items) {
          settings.value[item.key] = item.value
        }
      }

      return { settings, loading, save, refresh }
    }
    ```

    Design decisions (Claude's Discretion):
    - `category` scopes the useState key (`admin-settings-${category}`) to prevent cache collisions between admin pages that use different categories.
    - `refresh()` fetches ALL settings (not filtered by category) because the API returns all settings anyway. Filtering by category would require a separate API endpoint.
    - `save()` auto-populates `category` from the constructor parameter when items don't specify their own — saves boilerplate in admin page code.
    - `save()` updates the local `settings` ref after successful API response — UI reflects changes immediately without re-fetching.
    - File location: `composables/admin/useAdminSettings.ts` per D-07.
    - The `save(values)` parameter uses the same shape as SettingsService.batchUpdate() items: `{ key, value, category?, description? }`.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client composable → /api/settings GET | Public endpoint — settings data crosses from server to client. No auth required for GET (public data) |
| Client composable → /api/settings PUT | Protected endpoint — admin auth required. The composable sends data; auth is enforced server-side in index.put.ts |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-04-01 | Information Disclosure | useSiteSettings composable | accept | Reads from GET /api/settings (public endpoint). Returns site config which is public data (site name, social links, etc.) — no PII or secrets |
| T-01-04-02 | Tampering | useSiteSettings.refresh() | accept | Fetching from a public GET endpoint — no write capability. Data tampering at the API layer is mitigated by Zod validation in Phase 1 Plan 01 |
| T-01-04-03 | Elevation of Privilege | useAdminSettings.save() | mitigate | Auth enforced server-side at PUT /api/settings via `event.context.user` check. Client-side composable has no auth bypass — sends standard HTTP request that server validates |
| T-01-04-04 | Spoofing | useState key collision | mitigate | useAdminSettings uses unique useState key `admin-settings-${category}` — category-scoped keys prevent cross-page state pollution |
</threat_model>
