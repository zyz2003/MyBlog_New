    ```typescript
    /**
     * Save theme config with dual-write to system_settings and theme_settings (per D-02, D-03)
     * Wraps both writes in a single transaction for atomicity
     */
    async saveConfig(themeName: string, config: ThemeConfig): Promise<void> {
      // Update in-memory manifest if theme is loaded
      const manifest = this.themes.get(themeName)
      if (manifest) {
        manifest.config = config
      }

      // Atomic dual-write: system_settings.themeConfig + theme_settings.config
      await db.transaction(async (tx) => {
        // 1. Sync to system_settings for unified config (per D-02)
        // Import SettingsService dynamically to avoid issues with transaction context
        // But since SQLite transactions are connection-level, module-level db works within db.transaction()
        // We use SettingsService directly — the transaction context is connection-level
        await SettingsService.upsert('themeConfig', config, 'theme', 'Active theme configuration')

        // 2. Upsert into theme_settings table
        const existing = await tx
          .select()
          .from(themeSettings)
          .where(eq(themeSettings.themeName, themeName))
          .limit(1)

        if (existing.length > 0) {
          await tx
            .update(themeSettings)
            .set({ config, updatedAt: new Date() })
            .where(eq(themeSettings.themeName, themeName))
        } else {
          await tx
            .insert(themeSettings)
            .values({ themeName, config, isActive: false })
        }
      })
    }
    ```

    ```typescript
    /**
     * Async helper: load a theme manifest from the theme_settings DB table
     * Stores result in this.themes Map on success
     */
    private async loadThemeFromDb(themeName: string): Promise<ThemeManifest | null> {
      try {
        const rows = await db
          .select()
          .from(themeSettings)
          .where(eq(themeSettings.themeName, themeName))
          .limit(1)

        if (rows.length === 0 || !rows[0].config) return null

        const row = rows[0]
        const config = row.config as ThemeConfig
        const manifest: ThemeManifest = {
          meta: {
            name: row.themeName,
            version: '1.0.0',
            description: `Theme "${row.themeName}" from database`,
          },
          config,
        }

        this.themes.set(themeName, manifest)

        hookEmitter.dispatch('theme:loaded', {
          themeName,
          config,
          timestamp: Date.now(),
        })

        return manifest
      } catch (error) {
        console.warn(`[ThemeManager] Failed to load theme "${themeName}" from DB:`, error)
        return null
      }
    }
    ```

    ```typescript
    import { themeManager } from '../../../core/theme'
    import { successResponse, errorResponse, AuthErrors, ValidationErrors } from '../../../utils/response'

    /**
     * POST /api/themes/:name/config
     * Protected endpoint — saves theme configuration via ThemeManager.saveConfig()
     * Per D-03: Delegates to ThemeManager for dual-write (system_settings + theme_settings)
     * Per D-04: No longer writes to filesystem config.json
     */
    export default defineEventHandler(async (event) => {
      // Check auth — admin only
      if (!event.context.user) {
        throw createError({
          statusCode: 401,
          data: errorResponse(AuthErrors.UNAUTHENTICATED.code, AuthErrors.UNAUTHENTICATED.message),
        })
      }

      const name = getRouterParam(event, 'name')
      if (!name) {
        throw createError({
          statusCode: 400,
          data: errorResponse(ValidationErrors.MISSING_PARAM.code, 'Theme name is required'),
        })
      }

      // Validate theme name to prevent path traversal (kept for safety)
      if (!/^[a-zA-Z0-9_-]+$/.test(name)) {
        throw createError({
          statusCode: 400,
          data: errorResponse(ValidationErrors.INVALID_PARAM.code, 'Invalid theme name'),
        })
      }

      const body = await readBody<Partial<ThemeConfig>>(event)
      if (!body) {
        throw createError({
          statusCode: 400,
          data: errorResponse(ValidationErrors.MISSING_PARAM.code, 'Config is required'),
        })
      }

      // Build complete config from body (validate required top-level keys exist)
      if (!body.colors || !body.fonts || !body.spacing || !body.borderRadius || !body.layout) {
        throw createError({
          statusCode: 400,
          data: errorResponse(ValidationErrors.INVALID_PARAM.code,
            'Config must include colors, fonts, spacing, borderRadius, and layout'),
        })
      }

      const config: ThemeConfig = {
        colors: body.colors,
        fonts: body.fonts,
        spacing: body.spacing,
        borderRadius: body.borderRadius,
        layout: body.layout,
        shadows: body.shadows,
        transitions: body.transitions,
        components: body.components,
      }

      // Per D-03: Save via ThemeManager (dual-write to system_settings + theme_settings)
      await themeManager.saveConfig(name, config)

      return successResponse(config, `Theme "${name}" config saved`)
    })
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Admin API request → ThemeManager.saveConfig() | Theme config JSON crosses from authenticated admin request into dual-table DB writes |
| ThemeManager.scanThemes() → DB reads | Theme config data loaded from DB replaces filesystem config.json reads — eliminates path traversal risk from theme directory names |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-02-01 | Tampering | ThemeManager.saveConfig() | mitigate | db.transaction() wraps dual-write — if either system_settings or theme_settings write fails, both roll back. No partial config state |
| T-01-02-02 | Spoofing | config.post.ts theme name | mitigate | Theme name validated with regex `/^[a-zA-Z0-9_-]+$/` before passing to ThemeManager — prevents SQL injection via theme name (though Drizzle parameterizes queries, this is defense-in-depth) |
| T-01-02-03 | Information Disclosure | ThemeManager.loadThemeFromDb() | accept | Returns theme config from DB (public data — theme colors, fonts, spacing, etc.). No PII or secrets in theme configs |
| T-01-02-04 | Elevation of Privilege | config.post.ts | mitigate | Auth check `event.context.user` at endpoint entry — unauthenticated requests get 401 before reaching ThemeManager |
| T-01-02-05 | Tampering | Path traversal via theme name | mitigate | Removed filesystem config.json reads — loadTheme() now reads from DB, eliminating the path traversal vector from config.json file paths. Theme name regex validation kept as belt-and-suspenders |
</threat_model>
